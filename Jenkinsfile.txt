pipeline {
    agent any

    stages {

        stage('Checkout') {
            steps {
                echo 'Checking out code from GitHub...'
                checkout scm
            }
        }

        stage('Build') {
            steps {
                echo 'Building Maven application...'
                sh 'mvn clean package -DskipTests'
            }
        }

        stage('Test') {
            steps {
                echo 'Running Maven tests...'
                sh 'mvn test'
            }
        }

      stage('SonarQube Analysis') {
    steps {
        echo 'Running SonarQube analysis...'

        withSonarQubeEnv('sonarqube') {
            sh '''
                mvn sonar:sonar \
                -Dsonar.projectKey=scouter \
                -Dsonar.projectName=scouter
            '''
        }  
    }
}
    }

    post {
        success {
            echo '✅ Checkout, Build, Test and SonarQube completed successfully!'
        }

        failure {
            echo '❌ Pipeline failed!'
        }
    }
}
